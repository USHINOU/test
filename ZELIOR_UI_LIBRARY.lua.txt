local _BorderGradients = {}
local _BorderLoopStarted = false
local function _StartBorderLoop()
    if _BorderLoopStarted then return end
    _BorderLoopStarted = true
    RunService.Heartbeat:Connect(function(dt)
        local step = 80 * dt
        for i = #_BorderGradients, 1, -1 do
            local g = _BorderGradients[i]
            if g and g.Parent then
                g.Rotation = (g.Rotation + step) % 360
            else
                table.remove(_BorderGradients, i)
            end
        end
    end)
end

-- Responsive Mobile Scale
local function GetViewport()
    local cam = workspace.CurrentCamera
    return cam and cam.ViewportSize or Vector2.new(800, 450)
end

local function CalculateScale()
    local vp = GetViewport()
    local isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
    local baseW, baseH = 620, 400
    if isMobile or vp.X < 850 then
        local sx = (vp.X * 0.86) / baseW
        local sy = (vp.Y * 0.86) / baseH
        return math.clamp(math.min(sx, sy), 0.35, 0.82)
    end
    return 0.85
end

local function New(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    o.Parent = parent
    return o
end

local function Tween(obj, info, props)
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

local function Corner(parent, radius)
    return New("UICorner", { CornerRadius = UDim.new(0, radius) }, parent)
end

local function Label(parent, text, size, color, font)
    return New("TextLabel", {
        BackgroundTransparency = 1,
        Text = text,
        TextSize = size or 13,
        TextColor3 = color or TextPrimary,
        Font = font or Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center
    }, parent)
end

    if not name then return nil end
    if Icons[name] then return Icons[name] end
    local lower = string.lower(name)
    if Icons[lower] then return Icons[lower] end
    return nil
end

function FontManager:ApplyFont(name)
    local path = self.FontPaths[name]
    if not path or not self.RootGui then return end
    local weight = self.FontWeights[name] or Enum.FontWeight.Regular
    local style = self.FontStyles[name] or Enum.FontStyle.Normal
    local ok, newFont = pcall(function()
        return Font.new(path, weight, style)
    end)
    if not ok or not newFont then return end
    local scale = self.FontSizeScale[name] or 1.0

    local function apply(inst, depth)
        if depth > 15 then return end
        for _, ch in ipairs(inst:GetChildren()) do
            if ch:IsA("TextLabel") or ch:IsA("TextButton") or ch:IsA("TextBox") then
                pcall(function()
                    local baseSize = ch:GetAttribute("_baseTextSize")
                    if not baseSize then
                        baseSize = ch.TextSize
                        ch:SetAttribute("_baseTextSize", baseSize)
                    end
                    ch.FontFace = newFont
                    ch.TextSize = baseSize * scale
                end)
            end
            apply(ch, depth + 1)
        end
    end
    apply(self.RootGui, 0)
end

function FontManager:ApplyCustomFont(source)
    local ok, newFont = pcall(function()
        local src = tostring(source)
        if src:match("^rbxasset://") then
            return Font.new(src)
        elseif src:match("^rbxassetid://") then
            return Font.fromId(tonumber(src:match("%d+")))
        elseif tonumber(src) then
            return Font.fromId(tonumber(src))
        else
            return Font.new("rbxasset://fonts/families/"..src..".json")
        end
    end)
    if not ok or not newFont or not self.RootGui then return end
    local function apply(inst, depth)
        if depth > 15 then return end
        for _, ch in ipairs(inst:GetChildren()) do
            if ch:IsA("TextLabel") or ch:IsA("TextButton") or ch:IsA("TextBox") then
                pcall(function() ch.FontFace = newFont end)
            end
            apply(ch, depth + 1)
        end
    end
    apply(self.RootGui, 0)
end

-- الفكرة: أول صورة ثابتة تبقى، الفيديو ينفك على راحته حتى 20+ ثانية، وبعد يبدأ سلس 100%
do
    local AssetService = game:GetService("AssetService")
    local VIDEO_MANIFEST = "https://soft-ears-project.lovable.app/api/public/manifest/8zpsh9h2tg"
    local alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    local lookup = table.create(128, 0)
    for i = 1, #alphabet do lookup[string.byte(alphabet, i)] = i - 1 end
    
    local function byteReader(text)
        local total=#text; local i=1; local buf3=table.create(3,0); local bi=3
        return function()
            if bi<3 then bi+=1 return buf3[bi] end
            if i+3>total then return nil end
            local a,b,c,d=string.byte(text,i,i+3); i+=4
            local n=lookup[a]*262144+lookup[b]*4096+lookup[c]*64+lookup[d]
            buf3[1]=math.floor(n/65536); buf3[2]=math.floor(n/256)%256; buf3[3]=n%256
            bi=1; return buf3[1]
        end
    end
    
    local function decodeD444(text, prev, pixels)
        local nextByte=byteReader(text); local flag=nextByte()
        if not flag then return nil end
        local out=buffer.create(pixels*4)
        if prev then buffer.copy(out,0,prev,0,pixels*4) end
        if flag==1 then
            local p=0
            while p<pixels do
                local x,y,z=nextByte(), nextByte(), nextByte()
                if not x then break end
                local o=p*4
                buffer.writeu8(out,o, math.floor(x/16)*17)
                buffer.writeu8(out,o+1, (x%16)*17)
                buffer.writeu8(out,o+2, math.floor(y/16)*17)
                buffer.writeu8(out,o+3,255)
                p+=1
                if p<pixels then
                    local o2=p*4
                    buffer.writeu8(out,o2, (y%16)*17)
                    buffer.writeu8(out,o2+1, math.floor(z/16)*17)
                    buffer.writeu8(out,o2+2, (z%16)*17)
                    buffer.writeu8(out,o2+3,255)
                    p+=1
                end
            end
        else
            while true do
                local i1,i2,i3,rg,bb=nextByte(), nextByte(), nextByte(), nextByte(), nextByte()
                if not i1 then break end
                local o=(i1*65536+i2*256+i3)*4
                if o+3<pixels*4 then
                    buffer.writeu8(out,o, math.floor(rg/16)*17)
                    buffer.writeu8(out,o+1, (rg%16)*17)
                    buffer.writeu8(out,o+2, math.floor(bb/16)*17)
                    buffer.writeu8(out,o+3,255)
                end
            end
        end
        return out
    end

    task.spawn(function()
        local okMan, payload = pcall(function()
            local raw = game:HttpGet(VIDEO_MANIFEST)
            return HttpService:JSONDecode(raw)
        end)
        if not okMan or not payload then return end
        
        local vWidth, vHeight, vFps, vCount = payload.w, payload.h, payload.fps, payload.count
        local ORIGIN = string.match(VIDEO_MANIFEST, "^(https?://[^/]+)") or ""
        local SLUG = string.match(VIDEO_MANIFEST, "/manifest/([%w%-]+)") or ""
        local PIX_BASE = ORIGIN .. "/api/public/pixels/" .. SLUG .. "/"
        
        local VideoBG = New("ImageLabel", {
            Name = "VideoBackground",
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            ImageTransparency = 1,
            ScaleType = Enum.ScaleType.Stretch,
            ZIndex = 0
        }, Main)
        Corner(VideoBG, 16)
        
        local vEditable = AssetService:CreateEditableImage({ Size = Vector2.new(vWidth, vHeight) })
        VideoBG.ImageContent = Content.fromObject(vEditable)
        local vSize = Vector2.new(vWidth, vHeight)
        local vOrigin = Vector2.zero
        local vPixels = vWidth * vHeight
        local vBuffers = {}
        
        local function fetchBody(index)
            for attempt=1,3 do
                if not Main.Parent then return nil end
                local ok, body = pcall(game.HttpGet, game, PIX_BASE..index)
                if ok and body and #body>10 then return body end
                task.wait(0.2*attempt)
            end
            return nil
        end
        
        -- 1. أول صورة تظهر فورا كصورة ثابتة
        local body0 = fetchBody(0)
        if body0 then
            local okDec, buf0 = pcall(decodeD444, body0, nil, vPixels)
            if okDec and buf0 then
                vBuffers[0] = buf0
                vEditable:WritePixelsBuffer(vOrigin, vSize, buf0)
                TweenService:Create(VideoBG, TweenInfo.new(0.6), {ImageTransparency = 0.45}):Play()
            end
        end
        
        -- 2. خلي الفيديو ينفك على راحته حتى لو 20+ ثانية (واحد بواحد، بدون ضغط)
        for i=1, vCount-1 do
            if not Main.Parent then break end
            local body = fetchBody(i)
            if body then
                local prev = vBuffers[i-1]
                local ok, decoded = pcall(decodeD444, body, prev, vPixels)
                if ok and decoded then
                    vBuffers[i] = decoded
                end
            end
            task.wait(0.01) -- نفس صغير جدا باش ما يعلقش روبلوكس
        end
        
        -- 3. بعد ما يكمل فك كامل، ابدأ الفيديو سلس 100% بدون تقطيع
        if not Main.Parent then return end
        
        local frameTime = 1/vFps
        local index = 0
        local nextTime = os.clock() + frameTime
        while Main.Parent do
            RunService.RenderStepped:Wait()
            if os.clock() >= nextTime then
                index = (index + 1) % vCount
                nextTime = os.clock() + frameTime
                if os.clock() - nextTime > frameTime*2 then nextTime = os.clock() + frameTime end
                local frame = vBuffers[index]
                if frame then
                    vEditable:WritePixelsBuffer(vOrigin, vSize, frame)
                end
            end
        end
    end)
end

local function DetectExecutor()
    local executorName = "Unknown"
    local success = false
    
    -- الطريقة الحديثة
    pcall(function()
        if identifyexecutor then
            executorName = identifyexecutor()
            success = true
        elseif getexecutorname then
            executorName = getexecutorname()
            success = true
        end
    end)
    
    if not success then
        pcall(function()
            if isdelta or (getgenv and getgenv().isDelta) then executorName = "Delta"
            elseif syn and syn.protect_gui then executorName = "Synapse"
            elseif fluxus then executorName = "Fluxus"
            elseif iscodex or Codex then executorName = "Codex"
            elseif KRNL_LOADED then executorName = "KRNL"
            elseif arceus then executorName = "Arceus X"
            elseif getgenv and getgenv().evon then executorName = "Evon"
            elseif is_sirhurt_closure then executorName = "SirHurt"
            elseif pebc_execute then executorName = "ProtoSmasher"
            elseif sentinel then executorName = "Sentinel"
            elseif wearedevs then executorName = "WeAreDevs"
            elseif _G and _G.Vega then executorName = "Vega X"
            elseif _G and _G.COMET then executorName = "Comet"
            elseif getgenv and getgenv().Valyse then executorName = "Valyse"
            else executorName = "Delta" -- افتراضي دلتا اذا ما عرف
            end
        end)
    end
    
    -- تنظيف الاسم
    if type(executorName) == "table" then
        executorName = executorName[1] or executorName.Name or "Delta"
    end
    return tostring(executorName)
end

local function CodeToFlagEmoji(code)
    local flag = ""
    pcall(function()
        local upper = string.upper(code)
        if #upper == 2 then
            local b1 = string.byte(upper:sub(1,1))
            local b2 = string.byte(upper:sub(2,2))
            if b1 >= 65 and b1 <= 90 and b2 >= 65 and b2 <= 90 then
                flag = utf8.char(0x1F1E6 + b1 - 65) .. utf8.char(0x1F1E6 + b2 - 65)
            end
        end
    end)
    if flag == "" then flag = "🏳️" end
    return flag, string.upper(code)
end

            return tostring(selected)

    local function calculateHeights()
        local count = #optionList
        if count == 0 then
            return 36, 44 + 6 + 36 + 6, 36
        end
        local visibleCount = math.min(count, maxVisibleItems)
        local containerH = visibleCount * (itemH + itemGap) + 6
        local fullCanvasH = count * (itemH + itemGap) + 8
        local cardH = 44 + 6 + containerH + 6
        return containerH, cardH, fullCanvasH
    end

        local cHeight, targetCardH, cCanvasH = calculateHeights()
        listContainer.Size = UDim2.new(1, -20, 0, cHeight)
        listContainer.CanvasSize = UDim2.new(0, 0, 0, cCanvasH)
        if isOpen then
            Tween(dropdownCard, Fast, { Size = UDim2.new(1, 0, 0, targetCardH) })

    end

    local function ToggleDropdown()
        PlayClickSound()
        isOpen = not isOpen
        local _, targetH, _ = calculateHeights()

        if multiSelect then

            end

        else
            return selected == opt

    end

        local btn = optionButtons[opt]

        btn.BackgroundTransparency = isSel and 0.2 or 0.6

        if lbl then
            lbl.TextColor3 = isSel and PurpleBright or TextPrimary
        end
    end

    local function MakeOption(opt)
        if optionButtons[opt] then return end
        table.insert(optionList, opt)
        local orderIndex = #optionList
        
        local isNoPlayers = (tostring(opt) == "No players")

        local optBtn = New("TextButton", {
            Name = "Opt_" .. tostring(opt),
            Size = UDim2.new(1, -4, 0, itemH),
            BackgroundColor3 = isNoPlayers and Color3.fromRGB(20, 20, 30) or (isOptionSelected(opt) and Color3.fromRGB(52, 28, 94) or Color3.fromRGB(24, 14, 44)),

            AutoButtonColor = not isNoPlayers,
            Active = not isNoPlayers,
            Selectable = not isNoPlayers,
            Text = "",
            LayoutOrder = orderIndex,
            ZIndex = 23

        optLabel.Size = UDim2.new(1, -20, 1, 0)
        optLabel.ZIndex = 24

            PlayClickSound()
            if tostring(opt) == "No players" then return end -- غير قابل للضغط
            if multiSelect then
                local foundIdx = nil
                for i, v in ipairs(selected) do
                    if v == opt then
                        foundIdx = i
                        break
                    end
                end

                if foundIdx then
                    table.remove(selected, foundIdx)
                    updateButtonVisual(opt)
                    selLabel.Text = getDisplayText()
                    if callback then pcall(callback, selected) end
                else
                    if #selected < maxSelect then
                        table.insert(selected, opt)
                        updateButtonVisual(opt)
                        selLabel.Text = getDisplayText()
                        if callback then pcall(callback, selected) end

                        if #selected >= maxSelect then
                            if isOpen then
                                ToggleDropdown()
                            end
                        end
                    end
                end
            else
                selected = opt
                selLabel.Text = tostring(opt)
                ToggleDropdown()
                for o, _ in pairs(optionButtons) do
                    updateButtonVisual(o)
                end
                if callback then pcall(callback, opt) end
            end
        end)

        optionButtons[opt] = optBtn
        updateDimensions()
    end

    local function RemoveOption(opt)
        local btn = optionButtons[opt]
        if not btn then return end
        btn:Destroy()
        optionButtons[opt] = nil
        for i, v in ipairs(optionList) do
            if v == opt then
                table.remove(optionList, i)
                break
            end
        end

        -- Reassign LayoutOrders
        for i, name in ipairs(optionList) do
            if optionButtons[name] then
                optionButtons[name].LayoutOrder = i
            end
        end

        updateDimensions()

        if multiSelect then
            for i = #selected, 1, -1 do
                if selected[i] == opt then
                    table.remove(selected, i)
                end
            end
            selLabel.Text = getDisplayText()
            if callback then pcall(callback, selected) end
        else
            if selected == opt then
                selected = optionList[1] or "None"
                selLabel.Text = tostring(selected)
                if callback then pcall(callback, selected) end
            end
        end
    end

    -- Initial population
    if options then
        for _, opt in ipairs(options) do
            MakeOption(opt)
        end
    end

    local controller = {}
    function controller:Add(option)
        if type(option) == "table" then
            for _, o in ipairs(option) do MakeOption(o) end
        else
            MakeOption(option)
        end
    end
    function controller:Remove(option) RemoveOption(option) end
    function controller:Set(option)
        if multiSelect then
            selected = type(option) == "table" and option or {option}
            selLabel.Text = getDisplayText()
            for o, _ in pairs(optionButtons) do updateButtonVisual(o) end
        else
            if optionButtons[option] then
                selected = option
                selLabel.Text = tostring(option)
                for o, _ in pairs(optionButtons) do updateButtonVisual(o) end
            end
        end
    end
    function controller:Get() return selected end
    function controller:Card() return dropdownCard end
    controller.CardInstance = dropdownCard

    return controller
end

        Font = Enum.Font.GothamMedium,
        TextXAlignment = Enum.TextXAlignment.Left,

        ZIndex = 22
    }, searchFrame)

    box:GetPropertyChangedSignal("Text"):Connect(function()
        local q = box.Text:lower():gsub("^%s+", ""):gsub("%s+$", "")
        if q == "" then
            for _, child in ipairs(page:GetChildren()) do
                if child:IsA("Frame") and child ~= searchFrame and not child:IsA("UIListLayout") and not child:IsA("UIPadding") then

                    end
                end

            end

    end)

    box.FocusLost:Connect(function()
        Tween(searchFrame, Fast, { BackgroundTransparency = 0.35 })
    end)

local CurrentZeliorSound = nil
local function ZeliorSmartPlay(songId)
    local rawId = tostring(songId):gsub("%D", "")
    if rawId == "" then return end

    pcall(function()
        if CurrentZeliorSound then
            CurrentZeliorSound:Stop()
            CurrentZeliorSound:Destroy()
            CurrentZeliorSound = nil
        end
        local sound = Instance.new("Sound")
        sound.Name = "ZeliorMusicTrack"
        sound.SoundId = "rbxassetid://" .. rawId
        sound.Volume = 0.8
        sound.Looped = true
        sound.Parent = (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")) or game:GetService("SoundService")
        sound:Play()
        CurrentZeliorSound = sound
    end)

    pcall(function()
        local RS = game:GetService("ReplicatedStorage")
        local RE = RS:FindFirstChild("RE") or RS:FindFirstChild("Events")
        if RE then
            local vehRemote = RE:FindFirstChild("1NoMoto1rVehicle1s") or RE:FindFirstChild("1Vehicl1eMusic1")
            if vehRemote and vehRemote:IsA("RemoteEvent") then
                vehRemote:FireServer("Music", rawId)
            end
        end
    end)
end
